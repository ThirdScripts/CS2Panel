cmake_minimum_required(VERSION 3.25)
project(CS2DropRecorder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(CS2DropRecorder WIN32 main.cpp ../../cpp/vision/image.cpp)
target_include_directories(CS2DropRecorder PRIVATE ../../cpp ../../multibox)
target_compile_definitions(CS2DropRecorder PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(CS2DropRecorder PRIVATE /utf-8 /EHsc /W4)
target_link_libraries(CS2DropRecorder PRIVATE Qt6::Core windowsapp d3d11 dxgi gdiplus ole32 shell32 user32 gdi32)
